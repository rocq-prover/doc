type signature_mismatch_error =
  1. | InductiveFieldExpected of Declarations.mutual_inductive_body
  2. | DefinitionFieldExpected
  3. | ModuleFieldExpected
  4. | ModuleTypeFieldExpected
  5. | NotConvertibleInductiveField of Names.Id.t * (Environ.env * Constr.types * Constr.types) option
  6. | NotConvertibleConstructorField of Names.Id.t * (Environ.env * Constr.types * Constr.types) option
  7. | NotConvertibleBodyField of (Environ.env * Constr.constr * Constr.constr) option
  8. | NotConvertibleTypeField of Environ.env * Constr.types * Constr.types
  9. | CumulativeStatusExpected of bool
  10. | PolymorphicStatusExpected of bool
  11. | NotSameConstructorNamesField of Names.Id.t array * Names.Id.t array
  12. | NotSameInductiveNameInBlockField of Names.Id.t * Names.Id.t
  13. | FiniteInductiveFieldExpected of bool
  14. | InductiveNumbersFieldExpected of {
    1. got : int;
    2. expected : int;
    }
  15. | InductiveParams of {
    1. env : Environ.env;
    2. got : Constr.rel_context;
    3. expected : Constr.rel_context;
    }
  16. | RecordFieldExpected of bool
  17. | RecordProjectionsExpected of {
    1. expected : Names.Name.t list;
    2. got : Names.Name.t list;
    }
  18. | NotEqualInductiveAliases of Names.MutInd.t * Names.MutInd.t
  19. | IncompatibleUniverses of {
    1. err : UGraph.univ_inconsistency;
    2. env : Environ.env;
    3. t1 : Constr.types;
    4. t2 : Constr.types;
    }
  20. | IncompatibleQualities of {
    1. err : QGraph.elimination_error;
    2. env : Environ.env;
    3. t1 : Constr.types;
    4. t2 : Constr.types;
    }
  21. | IncompatiblePolymorphism of Environ.env * Constr.types * Constr.types
  22. | IncompatibleUnivConstraints of {
    1. env : Environ.env;
    2. got : UVars.AbstractContext.t;
    3. expect : UVars.AbstractContext.t;
    }
  23. | IncompatibleVariance
  24. | NoRewriteRulesSubtyping
