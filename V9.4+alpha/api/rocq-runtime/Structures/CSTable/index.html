Module Structures.CSTable

Low level access to the Canonical Structure database

type entry = {
  1. projection : Names.GlobRef.t;
  2. value : ValuePattern.t;
  3. solution : Names.GlobRef.t;
}
val entries : unit -> entry list

all returns the list of tuples p ; v ; s Note: p s = v

val entries_for : Environ.env -> projection:Names.GlobRef.t -> entry list

entries_for p returns the list of canonical entries that have p as their FieldName
